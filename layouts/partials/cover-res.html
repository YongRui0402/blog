{{- /* 回傳處理過的封面圖資源（沒有就回空字串）。 用法：{{ $img := partial "cover-res.html" (dict "ctx" . "size" "thumb") }} 這支是**唯一**決定「封面從哪來」的地方 —— cover.html 與搜尋索引 都呼叫它。抄成兩份的話，改預設封面時一定只會改到一邊。 來源優先序：front matter 的 cover → 站台的 params.defaultCover。 傳 "fallback" false 就不套預設封面（只有真的指定過 cover 才回東西）。 og:image 需要這個 —— 每篇都回同一張通用照片的話，分享連結的預覽圖 全長一樣，還不如用有品牌的那張卡片。 */ -}} {{- $ctx := .ctx -}} {{- $size := default "thumb" .size -}} {{- $fallback := ne .fallback false -}} {{- $img := "" -}} {{- with $ctx.Params.cover -}} {{- $img = resources.Get . -}} {{- if not $img }}{{ $img = resources.Get (printf "photos/%s" .) }}{{ end -}} {{- end -}} {{- if and (not $img) $fallback -}} {{- with $ctx.Site.Params.defaultCover }}{{ $img = resources.Get . }}{{ end -}} {{- end -}} {{- $out := "" -}} {{- if $img -}} {{- $spec := cond (eq $size "hero") "1200x400 webp q80 Center" "640x400 webp q75 Center" -}} {{- $out = $img.Fill $spec -}} {{- end -}} {{- return $out -}}
